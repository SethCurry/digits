cmake_minimum_required(VERSION 3.24)
set(CMAKE_LIBRARY_ARCHITECTURE x64)
project(digits)
add_executable(${PROJECT_NAME} main.cpp )
add_subdirectory(lib/leap)
add_subdirectory(lib/digits)
add_subdirectory(lib/digitspy)
add_subdirectory(lib/json)
add_subdirectory(lib/cli11)

target_include_directories(${PROJECT_NAME} PUBLIC "${CMAKE_CURRENT_LIST_DIR}/include" "${CMAKE_CURRENT_LIST_DIR}/leap/include")
target_link_libraries(${PROJECT_NAME} leap)
target_link_libraries(${PROJECT_NAME} libdigits)
target_link_libraries(${PROJECT_NAME} libdigitspy)
target_link_libraries(libdigits leap)
target_link_libraries(libdigitspy leap)
target_link_libraries(libdigitspy libdigits)
target_link_libraries(libdigits json)
target_link_libraries(libdigits cli11)
